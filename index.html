library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity decode2_4 is
  Port (
    e : in  STD_LOGIC_VECTOR(1 downto 0);
    d : out STD_LOGIC_VECTOR(3 downto 0)
  );
end decode2_4;

architecture Behavioral of
              decode2_4 is
begin
  process(e)
  begin
    if (e="00") then
      d<="0001";
    else if (e="01") then
      d<="0010";
    else if (e="10") then
      d<="0100";
    else
      d<="1000";
    end if;
    end if;
    end if;
  end process;
end Behavioral;

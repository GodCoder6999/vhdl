library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity AND_GATE_2 is
  Port (
    a : in  STD_LOGIC;
    b : in  STD_LOGIC;
    Y : out STD_LOGIC
  );
end AND_GATE_2;

architecture Behavioral of
              AND_GATE_2 is
begin
  Y <= a and b;
end Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity decode2_4 is
  Port (
    e : in  STD_LOGIC_VECTOR(1 downto 0);
    d : out STD_LOGIC_VECTOR(3 downto 0)
  );
end decode2_4;

architecture Behavioral of
              decode2_4 is
begin
  process(e)
  begin
    if (e="00") then
      d<="0001";
    else if (e="01") then
      d<="0010";
    else if (e="10") then
      d<="0100";
    else
      d<="1000";
    end if;
    end if;
    end if;
  end process;
end Behavioral;
▸ Block Diagram — 2:4 Decoder
Input: e — STD_LOGIC_VECTOR(1 downto 0)  |  Output: d — STD_LOGIC_VECTOR(3 downto 0)
Architecture: Behavioral with process(e) sensitivity list
▸ Circuit / Logic Diagram — 2:4 Decoder
Structure: 2 inverters + 4 AND gates  |  Dashed = inverted input  |  Solid = direct input

END decode_t;

  COMPONENT decode2_4
    PORT(
      e : IN  std_logic_vector(1 downto 0);
      d : OUT std_logic_vector(3 downto 0)
    );
  END COMPONENT;

  --Inputs

  signal d : std_logic_vector(3 downto 0);

BEGIN

  uut: decode2_4 PORT MAP (
    e => e,
    d => d
  );

  -- Stimulus process
  stim_proc: process
  begin
    e(1 downto 0)<="00"; wait for 100 ns; -- d=0001
    e(1 downto 0)<="01"; wait for 100 ns; -- d=0010
    e(1 downto 0)<="10"; wait for 100 ns; -- d=0100
    e(1 downto 0)<="11"; wait for 100 ns; -- d=1000
    wait;
  end process;

END;
